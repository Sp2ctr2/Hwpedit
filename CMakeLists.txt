cmake_minimum_required(VERSION 3.25)
project(HwpEdit VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_AUTOMOC ON)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/bin)
option(HWPEDIT_SANITIZE "Enable address and undefined behavior sanitizers" OFF)
find_package(Qt6 6.4 REQUIRED COMPONENTS Core Gui Widgets Xml PrintSupport Concurrent Test)
find_package(ICU REQUIRED COMPONENTS uc i18n)
find_package(ZLIB REQUIRED)
find_package(PkgConfig REQUIRED)
pkg_check_modules(LIBZIP REQUIRED IMPORTED_TARGET libzip)
if(HWPEDIT_SANITIZE)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
endif()
add_library(hwpedit_core src/document/model.cpp src/input/unicode.cpp src/commands/editor.cpp)
target_include_directories(hwpedit_core PUBLIC include)
target_link_libraries(hwpedit_core PUBLIC Qt6::Core Qt6::Gui ICU::uc ICU::i18n)
target_compile_options(hwpedit_core PRIVATE -Wall -Wextra -Wpedantic)
include(CTest)
if(BUILD_TESTING)
  add_executable(hwpedit-tests tests/core.cpp)
  target_link_libraries(hwpedit-tests PRIVATE hwpedit_core Qt6::Test)
  add_test(NAME core COMMAND hwpedit-tests)
  set_tests_properties(core PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 60)
endif()
